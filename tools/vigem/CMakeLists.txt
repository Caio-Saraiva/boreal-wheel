cmake_minimum_required(VERSION 3.20.2)
project(boreal_wheel_vigem_dependency LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ViGEmClient_DLL ON CACHE BOOL "Build official client DLL" FORCE)
if(NOT VIGEM_SOURCE)
  message(FATAL_ERROR "Pass -DVIGEM_SOURCE=<official ViGEmClient source directory>")
endif()
add_subdirectory("${VIGEM_SOURCE}" upstream)
target_compile_definitions(ViGEmClientShared PRIVATE VIGEM_DYNAMIC VIGEM_EXPORTS)
