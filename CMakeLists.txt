cmake_minimum_required(VERSION 3.16)
project(boreal_wheel VERSION 0.3.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build x64: cmake -A x64")
  endif()
  add_executable(boreal-wheel src/main.c src/i18n.c src/mapping.c src/platform_windows.c)
  target_link_libraries(boreal-wheel PRIVATE ws2_32 dinput8 dxguid winmm setupapi)
  target_compile_definitions(boreal-wheel PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  find_package(Threads REQUIRED)
  add_executable(boreal-wheel src/main.c src/i18n.c src/mapping.c src/platform_linux.c)
  target_link_libraries(boreal-wheel PRIVATE Threads::Threads m)
endif()
if(MSVC)
  target_compile_options(boreal-wheel PRIVATE /W4 /O2 /utf-8)
else()
  target_compile_options(boreal-wheel PRIVATE -Wall -Wextra -Wpedantic -O2)
endif()
add_custom_command(TARGET boreal-wheel POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/web $<TARGET_FILE_DIR:boreal-wheel>/web)
include(CTest)
if(BUILD_TESTING)
  add_executable(test-mapping tests/test_mapping.c src/mapping.c)
  target_include_directories(test-mapping PRIVATE src)
  if(NOT WIN32)
    target_link_libraries(test-mapping PRIVATE m)
  endif()
  add_test(NAME mapping COMMAND test-mapping)
endif()
